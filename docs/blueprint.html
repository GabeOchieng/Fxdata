01 · System Overview

Platform at a Glance

KaziHub is a multi-sided marketplace platform connecting gig workers, property owners and independent sellers with consumers across Kenya. It operates four distinct market modules — Service Marketplace, Rentals, Footwear Commerce, and a Market Intelligence dashboard — unified under a single identity, shared authentication, and a WhatsApp-first transaction layer.

02 · Architecture

System Architecture

Three-tier architecture with a React/HTML frontend, FastAPI service layer, and a PostgreSQL + Redis data tier. Async task processing via Celery handles notifications, M-Pesa callbacks and bulk operations.

from pydantic import BaseModel, field_validator, model_validator
from typing import Literal
from uuid import UUID
from decimal import Decimal
import phonenumbers

class OrderItemCreate(BaseModel):
    item_id:    UUID
    item_type:  Literal["service", "rental", "product"]
    qty:        int = 1
    unit_price: Decimal
    size:       str | None = None
    color:      str | None = None
    package_id: UUID | None = None

    @field_validator("qty")
    @classmethod
    def qty_positive(cls, v: int) -> int:
        if v < 1:
            raise ValueError("Quantity must be at least 1")
        return v

class OrderCreate(BaseModel):
    type:             Literal["service", "rental", "product"]
    items:            list[OrderItemCreate]
    delivery_type:    Literal["pickup","cbd","nairobi","outside","upcountry"] | None = None
    delivery_address: str | None = None
    notes:            str | None = None

    @model_validator(mode="after")
    def delivery_address_required(self) -> "OrderCreate":
        if self.delivery_type in ("nairobi", "outside", "upcountry"):
            if not self.delivery_address:
                raise ValueError("delivery_address required for this delivery type")
        return self

class OrderItemOut(BaseModel):
    model_config = {"from_attributes": True}
    id:         UUID
    item_id:    UUID
    item_type:  str
    qty:        int
    unit_price: Decimal
    size:       str | None
    color:      str | None

class OrderOut(BaseModel):
    model_config = {"from_attributes": True}
    id:               UUID
    type:             str
    status:           str
    subtotal:         Decimal
    delivery_fee:     Decimal
    delivery_type:    str | None
    delivery_address: str | None
    items:            list[OrderItemOut]
    whatsapp_message: str  # pre-formatted WA order message

class PaymentInitiate(BaseModel):
    order_id:     UUID
    phone_number: str  # +254 format

    @field_validator("phone_number")
    @classmethod
    def validate_ke_phone(cls, v: str) -> str:
        try:
            parsed = phonenumbers.parse(v, "KE")
            if not phonenumbers.is_valid_number(parsed):
                raise ValueError()
            return phonenumbers.format_number(
                parsed, phonenumbers.PhoneNumberFormat.E164
            )
        except Exception:
            raise ValueError("Invalid Kenyan phone number")

class MpesaCallbackSchema(BaseModel):
    """Incoming M-Pesa STK Push callback"""

        return self.Body["stkCallback"]["ResultCode"]

@router.post("/register", response_model=dict, status_code=201)
async def register(payload: UserCreate, db: AsyncSession = Depends(get_session)):
    """Register new user — sends OTP to phone number."""
    existing = await db.execute(
        select(User).where(User.phone == payload.phone)
    )
    if existing.scalar_one_or_none():
        raise HTTPException(status.HTTP_409_CONFLICT, "Phone already registered")

    otp = str(secrets.randbelow(900000) + 100000)  # 6-digit OTP
    user = User(
        phone=payload.phone,
        name=payload.name,
        password_hash=get_password_hash(payload.password) if payload.password else None,
    )
    db.add(user)
    await db.commit()
    await send_otp_sms(payload.phone, otp)          # Africa's Talking
    # store OTP in Redis with 10-min TTL
    await redis.setex(f"otp:{payload.phone}", 600, otp)
    return {"message": "OTP sent to your phone", "phone": payload.phone}

@router.post("/verify-otp", response_model=Token)
async def verify_otp(phone: str, otp: str, db: AsyncSession = Depends(get_session)):
    """Verify OTP and issue JWT tokens."""
    stored = await redis.get(f"otp:{phone}")
    if not stored or stored.decode() != otp:
        raise HTTPException(status.HTTP_400_BAD_REQUEST, "Invalid or expired OTP")

    user = await get_user_by_phone(db, phone)
    user.is_verified = True
    await db.commit()
    await redis.delete(f"otp:{phone}")

    return Token(
        access_token=create_access_token(str(user.id)),
        refresh_token=create_refresh_token(str(user.id)),
        token_type="bearer",
        user=UserOut.model_validate(user),
    )

import httpx, base64, json
from datetime import datetime
from app.core.config import settings

DARAJA_BASE = "https://api.safaricom.co.ke"  # live
# DARAJA_BASE = "https://sandbox.safaricom.co.ke"  # sandbox

class MpesaService:
    """Safaricom Daraja API v2 — STK Push integration."""

    async def get_access_token(self) -> str:
        """OAuth token — cached in Redis for 55 minutes."""
        cached = await redis.get("mpesa:token")
        if cached:
            return cached.decode()

        creds = base64.b64encode(
            f"{settings.MPESA_CONSUMER_KEY}:{settings.MPESA_CONSUMER_SECRET}".encode()
        ).decode()
        async with httpx.AsyncClient() as client:
            resp = await client.get(
                f"{DARAJA_BASE}/oauth/v1/generate?grant_type=client_credentials",
                headers={"Authorization": f"Basic {creds}"}
            )
        token = resp.json()["access_token"]
        await redis.setex("mpesa:token", 3300, token)
        return token

    async def stk_push(
        self, phone: str, amount: int,
        order_id: str, description: str = "KaziHub Order"
    ) -> dict:
        """Initiate Lipa Na M-Pesa STK Push."""
        token = await self.get_access_token()
        timestamp = datetime.now().strftime("%Y%m%d%H%M%S")
        password = base64.b64encode(
            f"{settings.MPESA_SHORTCODE}{settings.MPESA_PASSKEY}{timestamp}".encode()
        ).decode()

        payload = {
            "BusinessShortCode": settings.MPESA_SHORTCODE,
            "Password": password,
            "Timestamp": timestamp,
            "TransactionType": "CustomerPayBillOnline",
            "Amount": amount,
            "PartyA": phone,
            "PartyB": settings.MPESA_SHORTCODE,
            "PhoneNumber": phone,
            "CallBackURL": f"{settings.BASE_URL}/api/v1/orders/mpesa-callback",
            "AccountReference": f"KH-{order_id[:8].upper()}",
            "TransactionDesc": description,
        }
        async with httpx.AsyncClient() as client:
            resp = await client.post(
                f"{DARAJA_BASE}/mpesa/stkpush/v1/processrequest",
                headers={"Authorization": f"Bearer {token}"},
                json=payload
            )
        result = resp.json()
        # store CheckoutRequestID for callback matching
        await redis.setex(
            f"mpesa:pending:{result['CheckoutRequestID']}", 300, order_id
        )
        return result

    async def handle_callback(self, payload: dict) -> bool:
        """Process STK Push callback from Safaricom."""
        body = payload["Body"]["stkCallback"]
        checkout_id = body["CheckoutRequestID"]
        result_code = body["ResultCode"]

        order_id = await redis.get(f"mpesa:pending:{checkout_id}")
        if not order_id:
            return False

        if result_code == 0:  # Success
            items = body["CallbackMetadata"]["Item"]
            mpesa_ref = next(
                (i["Value"] for i in items if i["Name"] == "MpesaReceiptNumber"), None
            )
            await update_payment_success(order_id.decode(), mpesa_ref)
            await send_order_confirmation.delay(order_id.decode())  # Celery
        else:
            await update_payment_failed(order_id.decode(), result_code)

        await redis.delete(f"mpesa:pending:{checkout_id}")
        return True

import httpx
from app.core.config import settings

WA_API = "https://graph.facebook.com/v18.0"

class WhatsAppService:
    """WhatsApp Business Cloud API — messages and order notifications."""

    async def send_message(self, to: str, text: str) -> dict:
        """Send plain text message."""
        async with httpx.AsyncClient() as client:
            resp = await client.post(
                f"{WA_API}/{settings.WA_PHONE_ID}/messages",
                headers={
                    "Authorization": f"Bearer {settings.WA_TOKEN}",
                    "Content-Type": "application/json",
                },
                json={
                    "messaging_product": "whatsapp",
                    "to": to,
                    "type": "text",
                    "text": {"body": text}
                }
            )
        return resp.json()

    def format_order_message(self, order: Order, seller_phone: str) -> str:
        """Format a complete order as a WhatsApp-ready message."""
        lines = [
            "🛍️ *NEW ORDER — KaziHub*",
            "─────────────────────",
        ]
        for i, item in enumerate(order.items, 1):
            lines.append(f"{i}. *{item.name}*")
            if item.size:   lines.append(f"   Size: {item.size}")
            if item.color:  lines.append(f"   Colour: {item.color}")
            lines.append(f"   Qty: {item.qty} × KES {item.unit_price:,.0f}")
        lines += [
            "─────────────────────",
            f"Subtotal: KES {order.subtotal:,.0f}",
            f"Delivery: {'FREE' if order.delivery_fee == 0 else f'KES {order.delivery_fee:,.0f}'}",
            f"*TOTAL: KES {(order.subtotal + order.delivery_fee):,.0f}*",
        ]
        if order.delivery_address:
            lines.append(f"📍 Deliver to: {order.delivery_address}")
        if order.notes:
            lines.append(f"📝 Notes: {order.notes}")
        lines += [
            "─────────────────────",
            f"Order Ref: KH-{str(order.id)[:8].upper()}",
            "✅ Please confirm & share M-Pesa details.",
            "Thank you! 🙏",
        ]
        return "\n".join(lines)

    def get_wa_link(self, phone: str, message: str) -> str:
        """Generate wa.me link for frontend one-tap order."""
        import urllib.parse
        return f"https://wa.me/{phone}?text={urllib.parse.quote(message)}"

    async def send_order_to_seller(self, order: Order) -> None:
        """Send formatted order notification to seller via WA API."""
        seller = order.items[0].seller
        if not seller.profile or not seller.profile.whatsapp:
            return
        msg = self.format_order_message(order, seller.profile.whatsapp)
        await self.send_message(seller.profile.whatsapp, msg)
